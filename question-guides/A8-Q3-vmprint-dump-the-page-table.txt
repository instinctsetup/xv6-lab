xv6 OS LAB - ASSIGNMENT 8, A8 Q3: vmprint: dump the page table
================================================================

WHAT THE QUESTION ASKS
----------------------
A vmprint() syscall printing every valid page directory entry and its page table entries (index, virtual address, frame, P/W/U) below KERNBASE with two-level indentation, plus a summary line. Spot the stack guard page (PTE_U cleared) that exec() creates.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make qemu CPUS=1
At the xv6 shell: vmprinttest

WHICH FILES TO OPEN / SHOW
--------------------------
vm.c (vmprint, modelled on freevm's walk)

EXPECTED OUTPUT
---------------
A consistent dump at startup, new entries after sbrk(8 * 4096), one removed after sbrk(-4096), and the guard page identifiable.

CODE (verbatim from the repo)
---------------------------

----- vm.c - vmprint() -----
// Assignment 8 Q3 / Assignment 9 Q1: print the page table of the calling
// process as a two-level dump. For every present user PDE, print its
// index, the page table's physical address and flags; under it, indented,
// every present PTE with its VA, frame and flags. Pages mapped with
// PTE_U clear (the exec() stack guard page) are shown marked GUARD.
// Ends with the summary lines both questions ask for: total valid user
// pages, page-table pages used, and how many kernel PDE entries exist.
// Returns the number of valid user pages (the A9 Q1 test uses it).
int
vmprint(pde_t *pgdir)
{
  int i, j, n = 0, guard = 0, kentries = 0, ptpages = 0;
  cprintf("page table dump:\n");
  for(i = 0; i < NPDENTRIES; i++){
    pde_t pde = pgdir[i];
    pte_t *pgtab;
    if((pde & PTE_P) == 0)
      continue;
    if(i >= PDX(KERNBASE)){
      kentries++;              // kernel half of the address space
      continue;
    }
    cprintf("PDE [%d] -> pt at 0x%x [%s%s]\n", i, PTE_ADDR(pde),
            (pde & PTE_W) ? "W" : "-", (pde & PTE_U) ? "U" : "-");
    ptpages++;
    pgtab = (pte_t*)P2V(PTE_ADDR(pde));
    for(j = 0; j < NPTENTRIES; j++){
      pte_t pte = pgtab[j];
      uint va;
      if((pte & PTE_P) == 0)
        continue;
      va = (i << PDXSHIFT) | (j << PTXSHIFT);
      if((pte & PTE_U) == 0){
        cprintf("  .. PTE [%d] va 0x%x -> pa 0x%x [GUARD: PTE_U clear]\n",
                j, va, PTE_ADDR(pte));
        guard++;
        continue;
      }
      cprintf("  .. PTE [%d] va 0x%x -> pa 0x%x [%s%s]\n",
              j, va, PTE_ADDR(pte),
              (pte & PTE_W) ? "W" : "-", "U");
      n++;
    }
  }
  cprintf("page table dump: %d user pages, %d guard pages, %d page-table "
          "pages, %d kernel PDE entries\n", n, guard, ptpages, kentries);
  return n;
}

----- vmprinttest.c - complete test program -----
#include "types.h"
#include "stat.h"
#include "user.h"

#define PGSIZE 4096

int
main(int argc, char *argv[])
{
  printf(1, "========================================================\n");
  printf(1, "      xv6 Question 3: Process Page Table Dump Test      \n");
  printf(1, "========================================================\n\n");

  // Step 1: Startup page table dump
  printf(1, "[Phase 1] Process Page Table at Startup:\n");
  printf(1, "--------------------------------------------------------\n");
  vmprint();
  printf(1, "--------------------------------------------------------\n");
  printf(1, "Stack Guard Page Analysis:\n");
  printf(1, "  - Notice PTE at index 2 (va = 0x2000): flags = [P, W, -]\n");
  printf(1, "  - User bit (PTE_U) is CLEARED by exec() via clearpteu()!\n");
  printf(1, "  - This prevents stack overflow from corrupting text/data segments.\n");
  printf(1, "  - User-accessible pages = (sz / PGSIZE) - 1 guard page.\n\n");

  // Step 2: Grow heap by 8 pages (32768 bytes)
  printf(1, "[Phase 2] Growing heap by 8 pages via sbrk(8 * 4096):\n");
  char *p = sbrk(8 * PGSIZE);
  if(p == (char*)-1){
    printf(1, "sbrk failed\n");
    exit();
  }
  for(int i = 0; i < 8; i++){
    p[i * PGSIZE] = 'A' + i; // Touch all 8 pages
  }
  printf(1, "--------------------------------------------------------\n");
  vmprint();
  printf(1, "--------------------------------------------------------\n");
  printf(1, "Observation: 8 new consecutive PTEs (index 4 to 11) appeared!\n\n");

  // Step 3: Shrink heap by 1 page (4096 bytes)
  printf(1, "[Phase 3] Shrinking heap by 1 page via sbrk(-4096):\n");
  sbrk(-1 * PGSIZE);
  printf(1, "--------------------------------------------------------\n");
  vmprint();
  printf(1, "--------------------------------------------------------\n");
  printf(1, "Observation: Top PTE (index 11) successfully disappeared!\n");
  printf(1, "Hierarchical dump accurately tracks dynamic heap life cycle.\n");
  printf(1, "========================================================\n");

  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Find the guard page in the output and count user pages against sz / PGSIZE. Run vmprinttest, screenshot all three dumps.

HOW IT IS IMPLEMENTED
---------------------
vmprint(): two nested loops modelled on freevm() - for each present page directory entry below KERNBASE print it, then print each present PTE under it with its virtual address, frame and flags, indented by level. The guard page is the one with PTE_U cleared.
