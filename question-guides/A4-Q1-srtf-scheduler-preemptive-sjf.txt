xv6 OS LAB - ASSIGNMENT 4, A4 Q1: SRTF scheduler (preemptive SJF)
================================================================

WHAT THE QUESTION ASKS
----------------------
Like SJF but preemptive: remaining_time starts at the declared burst, the timer tick in trap() decrements it, and the CPU goes to the RUNNABLE process with the smallest remaining time.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make clean && make SCHED=SRTF qemu CPUS=1
At the xv6 shell: srtftest

WHICH FILES TO OPEN / SHOW
--------------------------
proc.c (SRTF case in key(), sched_charge() decrements remaining_time and returns 1 = re-pick every tick), trap.c

EXPECTED OUTPUT
---------------
A newly arrived short job preempts the long one; the short job finishes first (verified).

  case SCHED_SRTF:                    // A4 Q1: shortest remaining time;
    // unset counts as the default burst, never ahead of a known short job.
    if(p->burst_time < 0)
      return 0x7fffffff;
    return (uint)(p->remaining_time >= 0 ? p->remaining_time : p->burst_time);

----- proc.c - sched_charge() (1 = re-pick every tick) -----
// Charge one tick of runtime and apply per-policy quantum accounting.
// Called from trap.c while handling a timer interrupt on a RUNNING proc.
// Returns 1 when the policy wants the CPU back (preemption).
int
sched_charge(struct proc *p)
{
  p->rtime++;
  p->qticks++;
  switch(SCHED){
  case SCHED_SRTF:                     // A4 Q1
    if(p->burst_time >= 0 && p->remaining_time > 0)
      p->remaining_time--;
    return 1;                          // re-pick every tick
  case SCHED_MLFQ: {                   // A4 Q3: demote at quantum expiry
    int limit = p->queue_level == 0 ? MLFQ_Q0_TICKS :
                p->queue_level == 1 ? MLFQ_Q1_TICKS : MLFQ_Q2_TICKS;
    if(p->qticks >= limit){
      if(p->queue_level < MLFQ_LEVELS - 1)
        p->queue_level++;
      p->qticks = 0;
    }
    return 1;
  }
  case SCHED_RR:
  case SCHED_PRIO:                     // A3 Q2/A7 Q4: preemptive, quantum 1
    return 1;
  case SCHED_FCFS:                     // A3 Q1: non-preemptive
  case SCHED_SJF:                      // A3 Q3: non-preemptive
  case SCHED_HRRN:                     // A4 Q2: non-preemptive
  default:
    return 0;
  }
}

// ---- Assignment 3 Q3 / Assignment 4: set_burst(int pid, int bt) ----
int
sys_set_burst(void)
{
  int pid, bt;
  if(argint(0, &pid) < 0 || argint(1, &bt) < 0 || bt <= 0)
    return -1;
  return set_proc_field(pid, bt, 0);
}

----- srtftest.c - complete test program -----
// Assignment 4, Question 1 test: SRTF (shortest remaining time first).
// Build and run with: make qemu SCHED=SRTF CPUS=1
// Expected: the late-arriving short job preempts the long job and
// finishes before it, unlike non-preemptive SJF.
#include "types.h"
#include "stat.h"
#include "user.h"
#include "pinfo.h"

static void work(int me, int burst)
{
  volatile int i, x = 0;
  for(i = 0; i < burst * 1000000; i++) x += i;
  struct pinfo pi;
  getpinfo(&pi);
  printf(1, "srtf: child %d done (burst %d, remaining %d)\n",
         me, burst, pi.remaining);
  exit();
}

int
main(void)
{
  int p0, p1;
  set_burst(getpid(), 1);   // keep the parent schedulable while setting up
  if((p0 = fork()) == 0) work(0, 90);   // long job starts first
  set_burst(p0, 90);
  sleep(5);                             // let it run, then...
  if((p1 = fork()) == 0) work(1, 10);   // ...a short job arrives
  set_burst(p1, 10);
  wait(); wait();
  printf(1, "srtf: test complete\n");
  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Compare sched_charge() for SRTF against SJF: returning 1 forces a yield and re-pick every tick. Run srtftest, screenshot the preemption.

HOW IT IS IMPLEMENTED
---------------------
Keep burst_time and add remaining_time, initialized by set_burst. On every timer tick in trap(), decrement the running process's remaining_time and yield, so the scheduler re-picks whoever now has the smallest remaining time.
