xv6 OS LAB - ASSIGNMENT 8, A8 Q4: Lazy allocation (demand paging)
================================================================

WHAT THE QUESTION ASKS
----------------------
sys_sbrk() only records growth; the T_PGFLT handler in trap.c allocates a zeroed frame on first touch and maps it PTE_W | PTE_U. Edge cases: kalloc failure, faults above sz, faults below the heap, kernel syscalls touching lazy buffers, and copyuvm() skipping never-allocated pages.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make clean && make LAZY=1 qemu CPUS=1
At the xv6 shell: lazytest

WHICH FILES TO OPEN / SHOW
--------------------------
sysproc.c (lazy sbrk), trap.c (T_PGFLT handler), vm.c (copyuvm skips absent pages), proc.h (heap_start)

EXPECTED OUTPUT
---------------
A 32 MB sbrk consumes 0 frames; touching 3 pages consumes exactly 3 (plus page tables); a read() into an untouched buffer works; an out-of-bounds write kills only that process (all 6 phases verified).

----- sysproc.c - sys_sbrk(), lazy grow only records the size -----
int
sys_sbrk(void)
{
  int addr;
  int n;

  if(argint(0, &n) < 0)
    return -1;
  addr = myproc()->sz;
#if LAZY_ALLOC
  // Assignment 8 Q4 / Assignment 9 Q2: lazy allocation. Growth only
  // records the new size; frames arrive on demand in trap.c's T_PGFLT
  // handler. Shrinking still frees the pages that were really allocated
  // (growproc's deallocuvm skips not-yet-backed lazy pages), per A8 Q4(a).
  if(n < 0){
    if((0U - (uint)n) > myproc()->sz - myproc()->heap_start)
      return -1;               // never shrink below the code/guard region
    return growproc(n) < 0 ? -1 : addr;
  }
  if((uint)n >= KERNBASE - myproc()->sz)
    return -1;                 // keep the user space below KERNBASE
  myproc()->sz += n;
  return addr;
#else
  if(growproc(n) < 0)
    return -1;
  return addr;
#endif
}

----- trap.c - page-fault handler allocates on demand -----
  case T_PGFLT:
    // Assignment 8 Q4 / Assignment 9 Q2: lazy allocation. sbrk() (see
    // sysproc.c) only records the new size; the first touch of each page
    // faults here and gets a frame on demand. This covers BOTH user-mode
    // touches and kernel-mode touches (a syscall copying to/from a lazy
    // user buffer) - the faulting address is what matters, not the mode.
    {
      struct proc *p = myproc();
      uint va;
      char *mem;
      if(p == 0)
        goto lazy_fail;
      va = rcr2();
      if(va < p->heap_start || va >= p->sz || uvmpresent(p->pgdir, va))
        goto lazy_fail;              // outside the recorded heap: real fault
      mem = kalloc();
      if(mem == 0){
        cprintf("lazyalloc: out of memory\n");
        goto lazy_fail;
      }
      memset(mem, 0, PGSIZE);
      if(mappages(p->pgdir, (char*)PGROUNDDOWN(va), PGSIZE, V2P(mem),
                  PTE_W|PTE_U) < 0){
        kfree(mem);
        goto lazy_fail;
      }
      p->pgfaults++;                 // A9 Q2: counted for getfaults()
      break;
    lazy_fail:
      if((tf->cs&3) == 0)
        panic("lazyalloc: kernel page fault outside user heap");
      cprintf("pid %d %s: unhandled page fault at 0x%x\n",
              p ? p->pid : 0, p ? p->name : "?", rcr2());
      if(p)
        p->killed = 1;
    }
    break;
#endif

----- vm.c - copyuvm() skips absent pages -----
// Given a parent process's page table, create a copy
// of it for a child.
pde_t*
copyuvm(pde_t *pgdir, uint sz)
{
  pde_t *d;
  pte_t *pte;
  uint pa, i, flags;
  char *mem;

  if((d = setupkvm()) == 0)
    return 0;
  for(i = 0; i < sz; i += PGSIZE){
    // consolidated build: lazy allocation (A8 Q4/A9 Q2) leaves pages in
    // [0,sz) without a PTE or frame; skip them instead of panicking.
    if((pte = walkpgdir(pgdir, (void *) i, 0)) == 0)
      continue;
    if(!(*pte & PTE_P))
      continue;
    pa = PTE_ADDR(*pte);
    flags = PTE_FLAGS(*pte);
    if((mem = kalloc()) == 0)
      goto bad;
    memmove(mem, (char*)P2V(pa), PGSIZE);
    if(mappages(d, (void*)i, PGSIZE, V2P(mem), flags) < 0) {
      kfree(mem);
      goto bad;
    }
  }
  return d;

bad:
  freevm(d);
  return 0;
}

----- lazytest.c - complete test program -----
#include "types.h"
#include "stat.h"
#include "user.h"
#include "meminfo.h"

#define PGSIZE 4096
#define BIG_SIZE (32 * 1024 * 1024) // 32 Megabytes = 8192 pages

int
main(int argc, char *argv[])
{
  struct meminfo mi_before, mi_after_sbrk, mi_after_touch;

  printf(1, "========================================================\n");
  printf(1, "     xv6 Question 4: Lazy Memory Allocation Test        \n");
  printf(1, "========================================================\n\n");

  getmeminfo(&mi_before);
  printf(1, "[Phase 1: Baseline Memory State]\n");
  printf(1, "  Free Physical Pages: %d (%d KB)\n", mi_before.free_pages, mi_before.free_pages * 4);
  printf(1, "  Process User Pages : %d (%d KB)\n\n", mi_before.user_pages, mi_before.user_pages * 4);

  // Task 4.d: Large 32 MB sbrk allocation without touching
  printf(1, "[Phase 2: Allocating 32 MB (8192 pages) via sbrk()]\n");
  char *heap = sbrk(BIG_SIZE);
  if(heap == (char*)-1){
    printf(1, "sbrk(32MB) failed!\n");
    exit();
  }

  getmeminfo(&mi_after_sbrk);
  printf(1, "  Returned Heap Base : 0x%x\n", (uint)heap);
  printf(1, "  Free Physical Pages: %d (Delta: %d pages)\n", 
         mi_after_sbrk.free_pages, mi_before.free_pages - mi_after_sbrk.free_pages);
  printf(1, "  Process User Pages : %d (Delta: +%d pages)\n",
         mi_after_sbrk.user_pages, mi_after_sbrk.user_pages - mi_before.user_pages);

  if(mi_before.free_pages - mi_after_sbrk.free_pages < 10){
    printf(1, "  -> [PASSED] 32 MB allocated in O(1) without consuming 8192 frames!\n\n");
  } else {
    printf(1, "  -> [FAILED] Memory was allocated eagerly!\n\n");
  }

  // Task 4.d: Touching only a few widely spaced pages
  printf(1, "[Phase 3: Touching 3 Widely Spaced Pages (Demand Paging)]\n");
  printf(1, "  Accessing page 0 (offset 0 MB)...\n");
  heap[0] = 'X';

  printf(1, "  Accessing page 1024 (offset 4 MB)...\n");
  heap[1024 * PGSIZE] = 'Y';

  printf(1, "  Accessing page 8000 (offset 31.25 MB)...\n");
  heap[8000 * PGSIZE] = 'Z';

  getmeminfo(&mi_after_touch);
  printf(1, "  Free Physical Pages After 3 Touches: %d\n", mi_after_touch.free_pages);
  printf(1, "  Consumed Physical Frames           : %d frames (Exact 3 pages + page tables)\n",
         mi_after_sbrk.free_pages - mi_after_touch.free_pages);
  printf(1, "  -> [PASSED] Only touched pages consumed physical memory frames!\n\n");

  // Task 4.c & 4.d: System call touching lazily allocated buffer
  printf(1, "[Phase 4: Kernel System Call on Untouched Lazy Buffer]\n");
  int pfd[2];
  if(pipe(pfd) < 0){
    printf(1, "pipe failed\n");
    exit();
  }
  // Untouched page at offset 8 MB (page 2048)
  char *untouched_buf = heap + 2048 * PGSIZE;
  write(pfd[1], "HELLO_LAZY_XV6", 14);
  read(pfd[0], untouched_buf, 14);
  untouched_buf[14] = '\0';
  printf(1, "  Read data into untouched buffer: '%s'\n", untouched_buf);
  printf(1, "  -> [PASSED] Kernel read/write transparently populated lazy page!\n\n");
  close(pfd[0]);
  close(pfd[1]);

  // Task 4.d: Testing fork() with lazy pages
  printf(1, "[Phase 5: Testing fork() with Lazy Pages]\n");
  int pid = fork();
  if(pid < 0){
    printf(1, "fork failed\n");
    exit();
  }
  if(pid == 0){
    // Child verifies reading from touched page
    if(heap[0] != 'X' || heap[1024 * PGSIZE] != 'Y'){
      printf(1, "Child read incorrect data!\n");
    } else {
      printf(1, "  [CHILD] Correctly read parent data from demand-paged heap.\n");
    }
    exit();
  }
  wait();
  printf(1, "  [PARENT] Child safely completed. copyuvm() skipped untouched pages.\n\n");

  // Task 4.c: Edge case - Invalid access triggers clean process termination
  printf(1, "[Phase 6: Edge Case - Out-of-Bounds Access Termination]\n");
  int pid2 = fork();
  if(pid2 == 0){
    printf(1, "  [CHILD 2] Attempting invalid write beyond sz (addr = 0x%x)...\n", (uint)(heap + BIG_SIZE + 4096));
    char *bad_addr = heap + BIG_SIZE + 4096;
    *bad_addr = '!'; // Triggers segmentation fault
    printf(1, "  [CHILD 2] Should not reach here!\n");
    exit();
  }
  wait();
  printf(1, "  [PARENT] Child 2 was cleanly terminated without crashing kernel!\n\n");

  printf(1, "========================================================\n");
  printf(1, "  Question 4 Lazy Allocation Verification: ALL PASSED!   \n");
  printf(1, "========================================================\n");
  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Follow one fault from rcr2() to mappages(). Note heap_start: faults below the heap kill the process instead of mapping page zero. Run lazytest, screenshot each phase.

HOW IT IS IMPLEMENTED
---------------------
sys_sbrk(): for n > 0 just do sz += n and return. trap() case T_PGFLT: read the faulting address with rcr2(); if it lies in [heap_start, sz) and has no present PTE, kalloc() a frame, zero it, mappages() it PTE_W|PTE_U and return from the trap; anything else kills the process. copyuvm() and deallocuvm() must skip never-allocated pages instead of panicking.
