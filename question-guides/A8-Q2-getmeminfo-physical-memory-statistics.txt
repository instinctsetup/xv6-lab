xv6 OS LAB - ASSIGNMENT 8, A8 Q2: getmeminfo: physical memory statistics
================================================================

WHAT THE QUESTION ASKS
----------------------
Track free pages in kalloc.c and fill a meminfo struct: total_pages, free_pages, the caller's user pages (sz / PGSIZE) and its page-table pages. Prove there is no leak: free pages return to the earlier value after a child exits and is reaped.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make qemu CPUS=1
At the xv6 shell: meminfo_test

WHICH FILES TO OPEN / SHOW
--------------------------
kalloc.c (kfree_pages), vm.c (count_pt_pages), sysproc.c (getmeminfo), meminfo.h

EXPECTED OUTPUT
---------------
+4 user pages and -4 free pages after sbrk, and zero leak after the child is reaped (verified: exact match, PASSED).

CODE (verbatim from the repo)
---------------------------

----- kalloc.c - kfree_pages() -----
// Assignment 10, Q1(b): number of pages currently sitting on the free
// list - the exact count of available physical memory pages.
int
kfree_pages(void)
{
  struct run *r;
  int n = 0;

  if(kmem.use_lock)
    acquire(&kmem.lock);
  for(r = kmem.freelist; r; r = r->next)
    n++;
  if(kmem.use_lock)
    release(&kmem.lock);
  return n;
}

----- vm.c - count_pt_pages() -----
// Assignment 8 Q2: pages consumed by a page directory and its page tables.
int
count_pt_pages(pde_t *pgdir)
{
  int i, n = 1;   // the page directory itself
  for(i = 0; i < NPDENTRIES; i++)
    if((pgdir[i] & PTE_P) && (pgdir[i] & PTE_U))
      n++;
  return n;
}

----- sysproc.c - sys_getmeminfo() -----
// ---- Assignment 8 Q2: physical memory statistics ----
int
sys_getmeminfo(void)
{
  struct meminfo *mi;
  struct proc *p = myproc();
  if(argptr(0, (void*)&mi, sizeof(*mi)) < 0)
    return -1;
  mi->total_pages = total_frames_count();          // kalloc.c
  mi->free_pages = kfree_pages();                  // kalloc.c (oslab10)
  mi->user_pages = p->sz / PGSIZE;
  mi->pt_pages = count_pt_pages(p->pgdir);         // vm.c
  return 0;
}

----- meminfo.h - shared struct (whole file) -----
#ifndef _MEMINFO_H_
#define _MEMINFO_H_

// Assignment 8, Question 2: structure filled by the getmeminfo() syscall
// with physical-memory statistics.
struct meminfo {
  int total_pages; // physical frames managed by the kernel allocator
  int free_pages;  // currently available free physical frames
  int user_pages;  // pages in the caller's user address space (sz / PGSIZE)
  int pt_pages;    // pages used by the caller's page directory + page tables
};

#endif

----- meminfo_test.c - complete test program -----
#include "types.h"
#include "stat.h"
#include "user.h"
#include "meminfo.h"

void print_mem_state(const char *phase, struct meminfo *mi)
{
  printf(1, "[%s]\n", phase);
  printf(1, "  Total System Pages : %d (%d MB)\n", mi->total_pages, (mi->total_pages * 4) / 1024);
  printf(1, "  Free Physical Pages: %d (%d KB)\n", mi->free_pages, mi->free_pages * 4);
  printf(1, "  Process User Pages : %d (%d KB)\n", mi->user_pages, mi->user_pages * 4);
  printf(1, "  Page Table Pages   : %d (%d KB)\n\n", mi->pt_pages, mi->pt_pages * 4);
}

int
main(int argc, char *argv[])
{
  struct meminfo mi_initial, mi_allocated, mi_fork, mi_final;

  printf(1, "========================================================\n");
  printf(1, "    xv6 Question 2: Physical Memory Statistics Test     \n");
  printf(1, "========================================================\n\n");

  // Step 1: Initial baseline memory state
  if(getmeminfo(&mi_initial) < 0){
    printf(1, "getmeminfo failed\n");
    exit();
  }
  print_mem_state("Phase 1: Initial Baseline", &mi_initial);

  // Step 2: Allocate memory via sbrk (4 pages = 16 KB)
  printf(1, "--- Growing heap by 4 pages (16384 bytes) via sbrk() ---\n");
  char *mem = sbrk(4 * 4096);
  if(mem == (char*)-1){
    printf(1, "sbrk failed\n");
    exit();
  }
  // Touch pages to ensure physical allocation
  for(int i = 0; i < 4; i++){
    mem[i * 4096] = (char)(i + 1);
  }

  if(getmeminfo(&mi_allocated) < 0){
    printf(1, "getmeminfo failed\n");
    exit();
  }
  print_mem_state("Phase 2: After sbrk(4 * 4096)", &mi_allocated);

  printf(1, "Allocation Verification:\n");
  printf(1, "  User pages delta : +%d pages (Expected: +4)\n", mi_allocated.user_pages - mi_initial.user_pages);
  printf(1, "  Free pages delta : -%d pages (Expected: -4)\n\n", mi_initial.free_pages - mi_allocated.free_pages);

  // Step 3: Fork a child process and examine memory consumption
  printf(1, "--- Forking child process to observe free_pages change ---\n");
  int pid = fork();
  if(pid < 0){
    printf(1, "fork failed\n");
    exit();
  }

  if(pid == 0){
    // In child process: copyuvm() cloned all user frames and page tables
    struct meminfo mi_child;
    getmeminfo(&mi_child);
    printf(1, "  [CHILD PID %d] Running:\n", getpid());
    printf(1, "    Free Physical Pages : %d\n", mi_child.free_pages);
    printf(1, "    Process User Pages  : %d\n", mi_child.user_pages);
    printf(1, "    Page Table Pages    : %d\n", mi_child.pt_pages);
    exit();
  }

  // In parent: inspect while child exists / reap child
  getmeminfo(&mi_fork);
  wait(); // Child exits and is reaped

  // Step 4: After child exits and is reaped
  if(getmeminfo(&mi_final) < 0){
    printf(1, "getmeminfo failed\n");
    exit();
  }
  print_mem_state("Phase 3: After Child Exits and is Reaped", &mi_final);

  // Step 5: Memory Leak Verification
  printf(1, "--------------------------------------------------------\n");
  printf(1, "Memory Leak Verification:\n");
  printf(1, "  Pre-fork free pages : %d\n", mi_allocated.free_pages);
  printf(1, "  Post-reap free pages: %d\n", mi_final.free_pages);
  
  if(mi_final.free_pages == mi_allocated.free_pages){
    printf(1, "  Delta               : 0 (Exact match!)\n");
    printf(1, "  Status              : [PASSED] ZERO MEMORY LEAK DETECTED!\n");
  } else {
    printf(1, "  Delta               : %d pages\n", mi_allocated.free_pages - mi_final.free_pages);
    printf(1, "  Status              : [FAILED] Memory leak observed!\n");
  }
  printf(1, "========================================================\n");

  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Read kfree_pages() and count_pt_pages(). Run meminfo_test, screenshot the leak check.

HOW IT IS IMPLEMENTED
---------------------
kfree_pages() walks the free list under kmem.lock and counts it; count_pt_pages() walks the page directory counting present page tables; getmeminfo() fills the struct with those plus total frames and sz / PGSIZE.
