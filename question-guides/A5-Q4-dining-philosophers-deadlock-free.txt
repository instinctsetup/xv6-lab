xv6 OS LAB - ASSIGNMENT 5, A5 Q4: Dining philosophers (deadlock-free)
================================================================

WHAT THE QUESTION ASKS
----------------------
Five philosophers, five forks as binary semaphores, cycles of thinking, hungry, eating. Must use a deadlock-avoidance strategy; this tree uses resource ordering (always pick the lower-numbered fork first).

HOW TO BUILD AND RUN
--------------------
Build and boot:   make qemu CPUS=1
At the xv6 shell: dining

WHICH FILES TO OPEN / SHOW
--------------------------
sem.c, dining.c (fork ordering per philosopher)

EXPECTED OUTPUT
---------------
All five philosophers complete their cycles with no hang and no starvation over the run.

CODE (verbatim from the repo)
---------------------------

----- sem.c - kernel semaphores (whole file) -----
// sem.c - Kernel counting semaphores.
//
// Assignment 7, Question 2: counting semaphores on top of xv6's
// sleep()/wakeup(), with the exact syscall surface the assignment asks for
// (sem_init / sem_wait / sem_signal over a fixed array of 10 semaphores).
// Assignment 5 (Questions 2-4) reuses the same semaphores for the
// producer-consumer, readers-writers and dining-philosophers programs.
//
// sem_destroy() is an extension beyond the assignment text: it frees a
// slot and wakes any waiters so a stuck program fails loudly instead of
// hanging forever.

#include "types.h"
#include "defs.h"
#include "param.h"
#include "spinlock.h"
#include "mmu.h"
#include "proc.h"

// ptable lives in proc.c; needed to index per-process hold counts.
extern struct {
  struct spinlock lock;
  struct proc proc[NPROC];
} ptable;

#define NSEM 10   // assignment 7 Q2(c): fixed-size array of semaphores

struct semaphore {
  int used;              // slot allocated by sem_init
  int value;             // current count
  struct spinlock lk;    // protects value; sleep() releases it while blocked
  int waiters;           // number of processes blocked on this semaphore
  int owner[NPROC];      // per-proc hold count (Assignment 7 Q3 deadlock aid)
};

static struct semaphore sems[NSEM];

int
sem_init_id(int id, int value)
{
  struct semaphore *s;

  if(id < 0 || id >= NSEM || value < 0)
    return -1;
  s = &sems[id];
  acquire(&s->lk);
  if(s->used){           // re-initialising a live semaphore is a caller bug
    release(&s->lk);
    return -1;
  }
  s->used = 1;
  s->value = value;
  s->waiters = 0;
  memset(s->owner, 0, sizeof(s->owner));
  release(&s->lk);
  return 0;
}

int
sem_wait_id(int id)
{
  struct semaphore *s;
  struct proc *p;

  if(id < 0 || id >= NSEM)
    return -1;
  s = &sems[id];
  p = myproc();
  acquire(&s->lk);
  if(!s->used){
    release(&s->lk);
    return -1;
  }
  // Block while the count is exhausted. sleep() releases s->lk while
  // blocked and re-acquires it before returning, so the value test stays
  // atomic. proc->killed breaks the loop if the process is killed.
  while(s->used && s->value == 0 && !(p && p->killed)){
    s->waiters++;
    if(p)
      p->waiting_on = id;   // Assignment 7 Q3: wait-for bookkeeping
    sleep(s, &s->lk);
    s->waiters--;
  }
  if(p)
    p->waiting_on = -1;
  if(!s->used){             // sem_destroy() woke us: fail, don't re-sleep
    release(&s->lk);
    return -1;
  }
  if(p && p->killed){
    release(&s->lk);
    return -1;
  }
  s->value--;
  if(p)
    s->owner[p - ptable.proc]++;
  release(&s->lk);
  return 0;
}

int
sem_signal_id(int id)
{
  struct semaphore *s;
  struct proc *p;

  if(id < 0 || id >= NSEM)
    return -1;
  s = &sems[id];
  p = myproc();
  acquire(&s->lk);
  if(!s->used){
    release(&s->lk);
    return -1;
  }
  if(p && s->owner[p - ptable.proc] > 0)
    s->owner[p - ptable.proc]--;
  s->value++;
  if(s->waiters > 0)
    wakeup(s);          // wakes every sleeper; they re-check value under lk
  release(&s->lk);
  return 0;
}

int
sem_destroy_id(int id)
{
  struct semaphore *s;

  if(id < 0 || id >= NSEM)
    return -1;
  s = &sems[id];
  acquire(&s->lk);
  if(!s->used){
    release(&s->lk);
    return -1;
  }
  s->used = 0;
  // Wake everyone blocked on the destroyed semaphore; they see used == 0
  // on re-check and fail instead of sleeping forever.
  wakeup(s);
  release(&s->lk);
  return 0;
}

----- dining.c - complete test program -----
// Assignment 5, Question 4: dining philosophers, deadlock-free by
// resource ordering (philosopher 4 picks up the right fork first).
// Forks are semaphores 5..9, one per philosopher.
// Expected: every philosopher eats MEALS times; the run terminates.
#include "types.h"
#include "stat.h"
#include "user.h"

#define NPHIL 5
#define MEALS 5
#define FORK_BASE 5

static void philosopher(int id)
{
  int left = FORK_BASE + id;
  int right = FORK_BASE + (id + 1) % NPHIL;
  for(int m = 0; m < MEALS; m++){
    // break the circular wait: the last philosopher reverses pickup order
    if(id == NPHIL - 1){
      sem_wait(right);
      sem_wait(left);
    } else {
      sem_wait(left);
      sem_wait(right);
    }
    printf(1, "dining: philosopher %d ate meal %d\n", id, m + 1);
    sem_signal(left);
    sem_signal(right);
  }
  exit();
}

int
main(void)
{
  for(int i = 0; i < NPHIL; i++)
    sem_init(FORK_BASE + i, 1);
  for(int i = 0; i < NPHIL; i++)
    if(fork() == 0)
      philosopher(i);
  for(int i = 0; i < NPHIL; i++)
    wait();
  printf(1, "dining: all philosophers finished - PASS\n");
  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Be ready to explain why picking the lower-numbered fork first breaks the circular wait. Run dining, screenshot the transitions.

HOW IT IS IMPLEMENTED
---------------------
One binary semaphore per fork, all starting at 1. Each philosopher picks up the lower-numbered of its two forks first, then the higher one, eats, and puts both down. That global ordering is what removes the circular wait.
