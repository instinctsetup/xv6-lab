xv6 OS LAB - ASSIGNMENT 4, A4 Q3: MLFQ scheduler (3 queues)
================================================================

WHAT THE QUESTION ASKS
----------------------
Three queues: level 0 runs 1 tick, level 1 runs 2, level 2 runs 4. Always pick from the highest non-empty queue, round-robin inside it, and demote a process that uses its whole quantum. New processes start at level 0.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make clean && make SCHED=MLFQ qemu CPUS=1
At the xv6 shell: mlfqtest

WHICH FILES TO OPEN / SHOW
--------------------------
proc.c (queue_level/qticks in proc.h, MLFQ case in pick(), quantum table in sched_charge(), periodic boost in sched_tick())

EXPECTED OUTPUT
---------------
A CPU-bound child sinks from queue 0 to queue 2 while a short job stays at the top and finishes fast (verified).

----- sched.h - MLFQ levels and quanta -----
//   HRRN - Assignment 4, Question 2: highest response ratio next
//   MLFQ - Assignment 4, Question 3: multilevel feedback queue
//
  ...
#define SCHED_HRRN  5   // A4 Q2
#define SCHED_MLFQ  6   // A4 Q3

  ...

// Assignment 4 Q3 (MLFQ): three queues. Ticks a process may run at each
// level before demotion, and the periodic boost interval that moves every
  ...
// process back to the top queue so low-priority work cannot starve.
#define MLFQ_LEVELS   3
#define MLFQ_Q0_TICKS 1
  ...
#define MLFQ_Q1_TICKS 2
#define MLFQ_Q2_TICKS 4
#define MLFQ_BOOST    100

----- proc.c - pick(), MLFQ walks queues top-down -----
static struct proc*
pick(void)
{
  struct proc *p, *best = 0;
  uint bkey = 0;

  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
    uint k;
    if(p->state != RUNNABLE)
      continue;
    if(SCHED == SCHED_MLFQ){
      // A4 Q3: never pick a lower queue while a higher queue has work.
      if(best != 0){
        if(p->queue_level > best->queue_level)
          continue;
        if(p->queue_level == best->queue_level &&
           p->last_run >= best->last_run)
          continue;
        best = p;
      } else {
        best = p;
      }
      continue;
    }
    k = key(p);
    if(best == 0 || k < bkey || (k == bkey && p->last_run < best->last_run)){
      best = p;
      bkey = k;
    }
  }
  return best;
}

----- proc.c - sched_charge(), quantum and demotion -----
// Charge one tick of runtime and apply per-policy quantum accounting.
// Called from trap.c while handling a timer interrupt on a RUNNING proc.
// Returns 1 when the policy wants the CPU back (preemption).
int
sched_charge(struct proc *p)
{
  p->rtime++;
  p->qticks++;
  switch(SCHED){
  case SCHED_SRTF:                     // A4 Q1
    if(p->burst_time >= 0 && p->remaining_time > 0)
      p->remaining_time--;
    return 1;                          // re-pick every tick
  case SCHED_MLFQ: {                   // A4 Q3: demote at quantum expiry
    int limit = p->queue_level == 0 ? MLFQ_Q0_TICKS :
                p->queue_level == 1 ? MLFQ_Q1_TICKS : MLFQ_Q2_TICKS;
    if(p->qticks >= limit){
      if(p->queue_level < MLFQ_LEVELS - 1)
        p->queue_level++;
      p->qticks = 0;
    }
    return 1;
  }
  case SCHED_RR:
  case SCHED_PRIO:                     // A3 Q2/A7 Q4: preemptive, quantum 1
    return 1;
  case SCHED_FCFS:                     // A3 Q1: non-preemptive
  case SCHED_SJF:                      // A3 Q3: non-preemptive
  case SCHED_HRRN:                     // A4 Q2: non-preemptive
  default:
    return 0;
  }
}

----- proc.c - sched_tick(), periodic boost -----
// Periodic bookkeeping for the preemptive dynamic policies.
// Called from trap.c on timer ticks with ptable.lock NOT held, so we take
// it here. Walking the table once per tick is O(NPROC) and fine for xv6.
void
sched_tick(void)
{
  struct proc *p;

  if(SCHED != SCHED_PRIO && SCHED != SCHED_MLFQ)
    return;
  acquire(&ptable.lock);
  for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
    if(SCHED == SCHED_PRIO){
      // A3 Q2/A7 Q4 aging: persist the effective-priority improvement so
      // long-waiting processes ratchet upward level by level.
      if(p->state == RUNNABLE && p->priority > 1 &&
         ticks - p->ready_since > AGING_INTERVAL){
        p->priority--;
        p->ready_since = ticks;
      }
    } else {
      // A4 Q3 MLFQ: periodic priority boost prevents starvation.
      if(ticks > 0 && ticks % MLFQ_BOOST == 0){
        p->queue_level = 0;
        p->qticks = 0;
      }
    }
  }
  release(&ptable.lock);
}

----- mlfqtest.c - complete test program -----
// Assignment 4, Question 3 test: MLFQ (3 queues, quantum 1/2/4 ticks,
// boost every MLFQ_BOOST ticks).
// Build and run with: make qemu SCHED=MLFQ CPUS=1
// Expected: the CPU-bound child sinks to queue 2 (queue_level grows in
// getpinfo samples) while the short child finishes quickly at the top;
// the periodic boost lifts the CPU-bound child back to queue 0.
#include "types.h"
#include "stat.h"
#include "user.h"
#include "pinfo.h"

static void work(int me)
{
  volatile int i, x = 0;
  struct pinfo pi;
  for(i = 0; i < 300000000; i++){
    x += i;
    if(i % 100000000 == 0){
      getpinfo(&pi);
      printf(1, "mlfq: child %d at queue %d (rtime %d)\n",
             me, pi.queue_level, pi.rtime);
    }
  }
  printf(1, "mlfq: child %d done\n", me);
  exit();
}

int
main(void)
{
  if(fork() == 0) work(0);       // CPU-bound
  if(fork() == 0){ sleep(2); printf(1, "mlfq: short child done\n"); exit(); }
  wait(); wait();
  printf(1, "mlfq: test complete\n");
  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Follow one process through sched_charge(): qticks reaches its queue limit, level rises, slice resets. Blocking voluntarily also ends the slice. Run mlfqtest, screenshot.

HOW IT IS IMPLEMENTED
---------------------
Add queue_level and qticks to proc.h. The scheduler scans queue 0, then 1, then 2, round-robin inside each. Every tick adds to qticks; at the queue limit (1, 2 or 4 ticks) demote one level and yield. Blocking voluntarily ends the slice, and a periodic boost returns everyone to queue 0.
