xv6 OS LAB - ASSIGNMENT 7, A7 Q1: clone() and join(): kernel threads
================================================================

WHAT THE QUESTION ASKS
----------------------
Add clone(fcn, arg, stack): a new process that shares the caller page table and starts at fcn(arg) on a caller-supplied stack, plus join() to reap it. The PDF pairs this with a deliberately unsynchronized shared counter so the race is visible before Q2 fixes it.

HOW TO BUILD AND RUN
--------------------
Build and boot:   make qemu CPUS=1
At the xv6 shell: threadtest

WHICH FILES TO OPEN / SHOW
--------------------------
proc.c (clone(), join(), thread-aware exit/wait/kill), proc.h (is_thread/leader/nthreads), umalloc.c (thread-safe malloc)

EXPECTED OUTPUT
---------------
Three threads run and join cleanly (verified: counter 300000 with synchronized increments). Without synchronization, lost updates are the lesson of the question.

----- proc.c - clone() -----
int
clone(void (*fcn)(void*), void *arg, void *stack)
{
  int i, pid;
  struct proc *np;
  struct proc *curproc = myproc();
  struct proc *leader = curproc->is_thread ? curproc->leader : curproc;

  if(fcn == 0 || stack == 0)
    return -1;
  // The kernel writes the fake frame on the user stack below; reject a
  // stack outside the address space instead of faulting in kernel mode.
  if((uint)stack > curproc->sz || curproc->sz - (uint)stack < PGSIZE ||
     (uint)fcn >= curproc->sz)
    return -1;

  if((np = allocproc()) == 0)
    return -1;

  // Share the caller's page table instead of copyuvm().
  np->pgdir = curproc->pgdir;
  np->sz = curproc->sz;
  np->heap_start = curproc->heap_start; // A8 Q4: inherit lazy heap base
  np->parent = curproc;
  *np->tf = *curproc->tf;

  // Inherit scheduler parameters like fork().
  np->priority = curproc->priority;
  np->burst_time = curproc->burst_time;
  np->remaining_time = curproc->remaining_time;

  // Start the thread at fcn(arg) on the supplied user stack.
  // Layout (x86 cdecl): [ret][arg] with ret = 0; a thread that returns
  // without calling exit() faults at address 0 and is killed loudly
  // rather than corrupting anything.
  np->tf->eip = (uint)fcn;
  np->tf->esp = (uint)stack + PGSIZE - 8;
  *((uint*)(np->tf->esp))     = 0;              // fake return address
  *((uint*)(np->tf->esp + 4)) = (uint)arg;
  np->tf->ebp = np->tf->esp;

  // Threads share open files and cwd by reference.
  for(i = 0; i < NOFILE; i++)
    if(curproc->ofile[i])
      np->ofile[i] = filedup(curproc->ofile[i]);
  np->cwd = idup(curproc->cwd);
  safestrcpy(np->name, curproc->name, sizeof(curproc->name));

  np->is_thread = 1;
  np->leader = leader;
  pid = np->pid;

  acquire(&ptable.lock);
  leader->nthreads++;
  np->state = RUNNABLE;
  release(&ptable.lock);

  return pid;
}

----- proc.c - join() -----
// returns the reaped thread's pid. PDF signature: int join(void).
int
join(void)
{
  struct proc *p;
  struct proc *curproc = myproc();
  struct proc *leader = curproc->is_thread ? curproc->leader : curproc;

  acquire(&ptable.lock);
  for(;;){
    int found = 0;
    for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
      if(!p->is_thread || p->leader != leader || p->state == UNUSED)
        continue;
      found = 1;
      if(p->state == ZOMBIE){
        int tid = p->pid;
        kfree(p->kstack);        // kernel stack only; the address space
        p->kstack = 0;           // and the user stack belong to the leader
        p->pid = 0;
        p->parent = 0;
        p->name[0] = 0;
        p->killed = 0;
        p->state = UNUSED;
        p->is_thread = 0;
        p->leader = p;
        leader->nthreads--;
        release(&ptable.lock);
        return tid;
      }
    }
    if(!found || curproc->killed){
      release(&ptable.lock);
      return -1;
    }
    sleep(leader, &ptable.lock);   // exit() wakeup1(leader) covers any joiner
  }
}

----- threadtest.c - complete test program -----
// Assignment 7, Question 1(e) test: clone()/join() kernel threads.
// PDF spec: create 3 threads with clone(), each incrementing a shared
// global counter 100,000 times, then join() each and print the counter.
// Threads share the address space, so the global counter is genuinely
// shared; a kernel semaphore (id 0) serializes the increments.
// Expected: final counter = 300000 -> PASS.
#include "types.h"
#include "stat.h"
#include "user.h"

#define NTHREADS 3
#define ITER 100000
#define SEM_MX 0

static int counter = 0;

static void
worker(void *arg)
{
  int me = *(int*)arg;
  for(int i = 0; i < ITER; i++){
    sem_wait(SEM_MX);
    counter++;
    sem_signal(SEM_MX);
  }
  printf(1, "threadtest: thread %d done\n", me);
  exit();
}

int
main(void)
{
  int ids[NTHREADS] = {0, 1, 2};
  sem_init(SEM_MX, 1);
  for(int i = 0; i < NTHREADS; i++){
    char *st = malloc(4096);
    if(!st){ printf(1, "threadtest: malloc failed\n"); exit(); }
    if(clone(worker, &ids[i], st) < 0){
      printf(1, "threadtest: clone %d failed\n", i);
      exit();
    }
  }
  for(int i = 0; i < NTHREADS; i++)
    printf(1, "threadtest: joined tid %d\n", join());
  printf(1, "threadtest: counter=%d expected=%d %s\n", counter,
         NTHREADS * ITER, counter == NTHREADS * ITER ? "PASS" : "FAIL");
  exit();
}

IN THE LAB: WHAT TO DO AND WHAT TO SHOW
---------------------------------------
Read clone(): the fake stack frame, the shared pgdir, and why exit() reaps a leader's threads before the address space is freed. Run threadtest, screenshot.

HOW IT IS IMPLEMENTED
---------------------
clone(): allocproc(), share the caller's pgdir instead of copyuvm(), copy the trapframe, set eip to the function and esp to the caller-supplied stack with a fake return frame on it. join(): find a ZOMBIE thread of this group and free its kernel stack only. exit(), wait() and kill() all need thread-aware paths so the shared address space is freed exactly once.
